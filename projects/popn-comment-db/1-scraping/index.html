import requests import json import os def dl_by_game(game_id): base_url = "https://web.archive.org/web/20130111034606/http://www.konami.jp/bemani/popn/music" + str(game_id) + "/mc/" songlist_filepath = "./songlists_json/" + str(game_id) + ".json" with open(songlist_filepath) as json_file: json_obj = json.load(json_file) songlist = json_obj['songlist'] headers = { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36" } for song in songlist: filename = "./songs_pm" + str(game_id) + "_raw/" + str(song['song_num']) + "_" + song['song_id'] + ".html" if not os.path.exists(filename): url = base_url + song['url'] response = requests.get(url, headers=headers) response.raise_for_status() response.encoding = response.apparent_encoding html_content = response.text with open(filename, "w", encoding="utf-8") as file: file.write(html_content) print("Successfully downloaded " + filename)
